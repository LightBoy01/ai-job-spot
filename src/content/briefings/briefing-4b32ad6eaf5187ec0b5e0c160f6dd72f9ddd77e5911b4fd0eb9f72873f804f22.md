<pre class="highlight plaintext"><code>SELECT table_name,
       TO_NUMBER(extractvalue(xmltype(
           dbms_xmlgen.getxml(
               'SELECT COUNT(*) c FROM ' || table_name)),
           '//C')) AS row_count
FROM user_tables
ORDER BY row_count DESC;
