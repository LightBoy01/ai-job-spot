import hashlib
import json
import datetime
import uuid

class GitHubProofOfSkill:
    def __init__(self, validator_id="did:ai-job-spot:validator-node-1"):
        self.validator_id = validator_id

    def _mock_github_api_response(self, username):
        """
        In production, this replaces requests.get(f'https://api.github.com/users/{username}/repos')
        Simulating a high-quality engineer's raw data.
        """
        return [
            {"name": "fast-api-scale", "stargazers_count": 1200, "language": "Python", "merged_prs": 45},
            {"name": "react-native-core", "stargazers_count": 3400, "language": "TypeScript", "merged_prs": 120},
            {"name": "legacy-java-system", "stargazers_count": 15, "language": "Java", "merged_prs": 5},
            {"name": "go-microservices", "stargazers_count": 890, "language": "Go", "merged_prs": 60}
        ]

    def generate_claim(self, username):
        # 1. Fetch Data
        raw_data = self._mock_github_api_response(username)
        
        # 2. Process Data (The "Logic" Layer)
        total_stars = sum(repo['stargazers_count'] for repo in raw_data)
        total_prs = sum(repo['merged_prs'] for repo in raw_data)
        languages = list(set(repo['language'] for repo in raw_data if repo['language']))
        
        # 3. Create Evidence Payload (The "Truth" Layer)
        evidence_payload = {
            "source": "github.com",
            "username_hash": hashlib.sha256(username.encode()).hexdigest(), # Privacy: Anonymize raw data
            "metrics": {
                "total_stars": total_stars,
                "verified_merged_prs": total_prs,
                "languages": languages
            },
            # In reality, this would be a Merkle Root or IPFS hash of the raw data
            "raw_data_snapshot_hash": hashlib.sha256(str(raw_data).encode()).hexdigest() 
        }
        
        # 4. Hash the Evidence (Cryptographic Proof)
        # This hash guarantees the data hasn't changed since verification
        evidence_string = json.dumps(evidence_payload, sort_keys=True).encode()
        snapshot_hash = hashlib.sha256(evidence_string).hexdigest()

        # 5. Construct the Verified Claim (The Output)
        claim = {
            "id": str(uuid.uuid4()),
            "timestamp": datetime.datetime.utcnow().isoformat() + "Z",
            "issuer": {
                "id": self.validator_id,
                "name": "AI Job Spot Verification Protocol"
            },
            "subject": {
                "id": f"user_did_{hashlib.md5(username.encode()).hexdigest()}",
            },
            "claim": {
                "category": "ENGINEERING_OUTPUT",
                "platform": "GITHUB",
                "assertion": f"Maintained repositories with {total_stars} collective stars and {total_prs} verified PRs.",
                "evidence": {
                    "snapshot_hash": snapshot_hash,
                    "data_summary": evidence_payload['metrics']
                }
            },
            "verification_status": {
                "verified": True,
                "confidence_score": 0.99,
                "method": "OAUTH_API_DIRECT_READ"
            }
        }
        
        return claim

# --- Execution ---
